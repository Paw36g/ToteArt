async function manejarEnvioResena(e) {
  if (e) e.preventDefault();

  const commentInput = document.getElementById('reviewCommentInput') || document.getElementById('comentarioResena');
  const commentText = commentInput ? commentInput.value.trim() : '';

  // Capturar estrellas seleccionadas
  let rating = 0;
  const starChecked = document.querySelector('input[name="rating_stars"]:checked');
  if (starChecked) {
    rating = parseInt(starChecked.value, 10);
  } else {
    const starPicker = document.querySelector('.rating-stars-picker');
    rating = starPicker ? parseInt(starPicker.dataset.selectedRating || '0', 10) : 0;
  }

  if (!rating || rating < 1 || rating > 5) {
    alert("Por favor selecciona una calificación de 1 a 5 estrellas antes de publicar.");
    return;
  }

  // Identificar usuario activo o autor
  const session = JSON.parse(localStorage.getItem('toteart_user') || '{}');
  const autor = session.nombres ? `${session.nombres} ${session.apellidos || ''}`.trim() :
                (session.nombre || 'Cliente ToteArt');

  const urlParams = new URLSearchParams(window.location.search);
  const currentProductId = urlParams.get('id') || window.CURRENT_PRODUCT_ID;

  if (!currentProductId) {
    console.error("No se localizó el identificador del producto actual.");
    return;
  }

  const payload = {
    id_producto: String(currentProductId),
    id_cliente: session.id_cliente || null,
    nombre_autor: autor,
    calificacion: rating,
    comentario: commentText
  };

  // 1. Guardar en Supabase
  let guardadoEnNube = false;
  if (window.supabaseClient) {
    try {
      const { error } = await window.supabaseClient.from('resenas').insert([payload]);
      if (!error) guardadoEnNube = true;
    } catch (err) {
      console.warn('[Reseñas] Fallback local por error en Supabase:', err);
    }
  }

  // 2. Respaldo local para persistencia inmediata
  try {
    const localKey = `toteart_reviews_${currentProductId}`;
    const localReviews = JSON.parse(localStorage.getItem(localKey) || '[]');
    localReviews.unshift({
      ...payload,
      fecha_resena: new Date().toISOString()
    });
    localStorage.setItem(localKey, JSON.stringify(localReviews));
  } catch (storageErr) {}

  alert("¡Gracias por tu valoración! Tu reseña ha sido registrada.");
  if (commentInput) commentInput.value = '';
  
  // Recalcular y renderizar de inmediato
  await cargarResenasYPromedio(currentProductId);
}

B. Recálculo de Promedio y Renderizado de Lista:

async function cargarResenasYPromedio(productId) {
  let reviews = [];

  // 1. Consultar Supabase
  if (window.supabaseClient) {
    try {
      const { data, error } = await window.supabaseClient
        .from('resenas')
        .select('*')
        .eq('id_producto', String(productId))
        .order('fecha_resena', { ascending: false });

      if (!error && Array.isArray(data) && data.length > 0) {
        reviews = data;
      }
    } catch (e) {}
  }

  // 2. Respaldo local si la base no devolvió datos
  if (reviews.length === 0) {
    try {
      const localKey = `toteart_reviews_${productId}`;
      reviews = JSON.parse(localStorage.getItem(localKey) || '[]');
    } catch(e) {}
  }

  // 3. Cálculo dinámico de promedio
  const total = reviews.length;
  let promedio = 5.0;
  if (total > 0) {
    const suma = reviews.reduce((acc, r) => acc + (Number(r.calificacion) || 0), 0);
    promedio = (suma / total).toFixed(1);
  }

  // 4. Actualizar textos de promedio en la cabecera del producto
  const avgTextElem = document.getElementById('productAverageRatingText') || document.querySelector('.product-rating-score');
  const countTextElem = document.getElementById('productReviewsCountText') || document.querySelector('.product-reviews-count');
  const starsElem = document.getElementById('productAverageStars') || document.querySelector('.product-stars-visual');

  if (avgTextElem) avgTextElem.textContent = `${promedio} / 5`;
  if (countTextElem) countTextElem.textContent = `(${total} reseña${total === 1 ? '' : 's'})`;
  if (starsElem) {
    const full = Math.round(Number(promedio));
    starsElem.innerHTML = '★'.repeat(full) + '☆'.repeat(5 - full);
    starsElem.style.color = '#F5A623';
  }

  // 5. Renderizar lista de comentarios y puntuaciones
  const container = document.getElementById('reviewsListContainer') || document.querySelector('.reviews-list');
  if (container) {
    if (total === 0) {
      container.innerHTML = `
        <div style="padding: 20px 0; color: #7a7672; font-style: italic;">
          Este tote bag aún no tiene opiniones. ¡Sé el primero en calificarlo!
        </div>
      `;
      return;
    }

    container.innerHTML = reviews.map(rev => {
      const fecha = new Date(rev.fecha_resena || Date.now()).toLocaleDateString('es-EC', {
        day: 'numeric', month: 'short', year: 'numeric'
      });
      const estrellas = '★'.repeat(rev.calificacion) + '☆'.repeat(5 - rev.calificacion);

      return `
        <div class="review-item" style="border-bottom: 1px solid #f0eee9; padding: 16px 0;">
          <div style="display: flex; justify-content: space-between; align-items: center; margin-bottom: 4px;">
            <strong style="color: #2E4A3D; font-size: 0.95rem;">${rev.nombre_autor || 'Cliente ToteArt'}</strong>
            <span style="font-size: 0.8rem; color: #9c9792;">${fecha}</span>
          </div>
          <div style="color: #F5A623; margin-bottom: 6px; font-size: 0.95rem;">${estrellas}</div>
          ${rev.comentario ? `<p style="margin: 0; color: #444; font-size: 0.9rem; line-height: 1.5;">${rev.comentario}</p>` : ''}
        </div>
      `;
    }).join('');
  }
}
